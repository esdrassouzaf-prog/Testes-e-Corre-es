<script>
  const STORAGE_KEY = 'realsi_apps_script_url';
  let dados = {
    insumos: [],
    historico: []
  };

  function getApiUrl() {
    // Se estiver rodando hospedado direto no Google Apps Script, usa a própria URL
    if (window.location.href.includes('script.google.com')) {
      return window.location.href.split('?')[0];
    }
    return localStorage.getItem(STORAGE_KEY) || '';
  }

  window.addEventListener('DOMContentLoaded', () => {
    const url = getApiUrl();
    if (!window.location.href.includes('script.google.com')) {
      document.getElementById('connBox').style.display = 'flex';
      document.getElementById('urlInput').value = url;
    }
    if (url) {
      carregarDados();
    } else {
      mostrarAlerta('Cole a URL do Apps Script acima para conectar com sua planilha.', 'err');
    }
  });

  function salvarUrl() {
    const val = document.getElementById('urlInput').value.trim();
    if (!val) return mostrarAlerta('Informe uma URL válida.', 'err');
    localStorage.setItem(STORAGE_KEY, val);
    carregarDados();
  }

  function mostrarAlerta(msg, tipo = 'ok') {
    const el = document.getElementById('alerta');
    el.textContent = msg;
    el.className = 'alert show ' + tipo;
    if (tipo === 'ok') {
      setTimeout(() => { el.className = 'alert'; }, 4000);
    }
  }

  async function carregarDados() {
    const url = getApiUrl();
    if (!url) return;
    try {
      const resp = await fetch(`${url}?action=getAll`);
      const json = await resp.json();
      if (!json.ok) throw new Error(json.error || 'Erro ao carregar dados');

      dados.insumos = json.insumos || [];
      dados.historico = json.historico || [];

      preencherInsumos();
      atualizarNumeroCorrecao();
      renderChipsLotes();
      renderHistorico();
    } catch (err) {
      mostrarAlerta('Erro ao conectar: ' + err.message, 'err');
    }
  }

  function preencherInsumos() {
    const sel = document.getElementById('fInsumo');
    const atual = sel.value;
    if (dados.insumos.length === 0) {
      sel.innerHTML = '<option value="">Cadastre insumos na aba Insumos da planilha</option>';
      return;
    }
    sel.innerHTML = '<option value="">Selecione o insumo...</option>' +
      dados.insumos.map(item => `<option value="${esc(item)}">${esc(item)}</option>`).join('');
    if (atual) sel.value = atual;
  }

  function atualizarNumeroCorrecao() {
    const lote = document.getElementById('fLote').value.trim().toLowerCase();
    if (!lote) {
      document.getElementById('fCorrecaoNum').value = '1ª Correção';
      return;
    }
    const qtdExistente = dados.historico.filter(h => h.lote.toLowerCase() === lote).length;
    document.getElementById('fCorrecaoNum').value = `${qtdExistente + 1}ª Correção`;
  }

  async function registrarCorrecao(e) {
    e.preventDefault();
    const url = getApiUrl();
    if (!url) return mostrarAlerta('Conecte a URL do Apps Script primeiro.', 'err');

    const btn = document.getElementById('btnSalvar');
    btn.disabled = true;
    btn.textContent = 'Salvando...';

    const lote = document.getElementById('fLote').value.trim();
    const misturador = document.getElementById('fMisturador').value;
    const insumo = document.getElementById('fInsumo').value;
    const quantidade = Number(document.getElementById('fQuantidade').value);
    const unidade = document.getElementById('fUnidade').value;
    const observacao = document.getElementById('fObs').value.trim();

    try {
      const resp = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify({
          action: 'salvarCorrecao',
          lote,
          misturador,
          insumo,
          quantidade,
          unidade,
          observacao
        })
      });
      const json = await resp.json();
      if (!json.ok) throw new Error(json.error || 'Erro ao salvar');

      dados.insumos = json.insumos || dados.insumos;
      dados.historico = json.historico || dados.historico;

      // Mantém o Lote e o Misturador preenchidos para facilitar lançar a próxima correção do mesmo lote
      document.getElementById('fQuantidade').value = '';
      document.getElementById('fObs').value = '';
      document.getElementById('pesquisaLote').value = lote;

      atualizarNumeroCorrecao();
      renderChipsLotes();
      renderHistorico();
      mostrarAlerta(`${json.numeroCorrecao}ª correção salva no Lote ${lote}!`, 'ok');
    } catch (err) {
      mostrarAlerta('Erro ao salvar: ' + err.message, 'err');
    } finally {
      btn.disabled = false;
      btn.textContent = 'Salvar Correção';
    }
  }

  function renderChipsLotes() {
    const container = document.getElementById('lotesRecentes');
    const filtroAtual = document.getElementById('pesquisaLote').value.trim().toLowerCase();
    const unicos = [];
    dados.historico.slice().reverse().forEach(item => {
      if (item.lote && !unicos.some(u => u.toLowerCase() === item.lote.toLowerCase())) {
        unicos.push(item.lote);
      }
    });

    if (unicos.length === 0) {
      container.innerHTML = '';
      return;
    }

    container.innerHTML =
      `<span class="chip ${filtroAtual === '' ? 'active' : ''}" onclick="filtrarPorLote('')">Todos</span>` +
      unicos.slice(0, 10).map(lote => {
        const ativo = filtroAtual === lote.toLowerCase() ? 'active' : '';
        return `<span class="chip ${ativo}" onclick="filtrarPorLote('${esc(lote)}')">Lote ${esc(lote)}</span>`;
      }).join('');
  }

  function filtrarPorLote(lote) {
    document.getElementById('pesquisaLote').value = lote;
    if (lote) {
      document.getElementById('fLote').value = lote;
      atualizarNumeroCorrecao();
    }
    renderChipsLotes();
    renderHistorico();
  }

  function renderHistorico() {
    const busca = document.getElementById('pesquisaLote').value.trim().toLowerCase();
    const tbody = document.getElementById('tbodyHistorico');
    const boxSoma = document.getElementById('boxSoma');
    const gridSoma = document.getElementById('gridSoma');

    const filtrados = dados.historico.filter(item =>
      !busca || item.lote.toLowerCase().includes(busca)
    );

    document.getElementById('contadorRegistros').textContent =
      `${filtrados.length} ${filtrados.length === 1 ? 'registro' : 'registros'}`;

    if (filtrados.length === 0) {
      tbody.innerHTML = `<tr><td colspan="3" class="empty">${busca ? 'Nenhuma correção encontrada para este lote.' : 'Nenhuma correção registrada ainda.'}</td></tr>`;
      boxSoma.style.display = 'none';
      return;
    }

    tbody.innerHTML = filtrados.map(item => `
      <tr>
        <td><span class="badge-num">${item.numeroCorrecao}</span></td>
        <td style="color:#475569; font-size:0.84rem;">${esc(item.dataHora)}</td>
        <td>
          <strong>Lote ${esc(item.lote)}</strong>
          <span class="badge-mist">${esc(item.misturador)}</span><br/>
          <span>${esc(item.descricao)}</span>
        </td>
      </tr>
    `).join('');

    // Se estiver filtrando um lote específico, mostra o somatório individual por insumo
    if (busca) {
      const somaPorInsumo = {};
      filtrados.forEach(item => {
        if (!somaPorInsumo[item.insumo]) somaPorInsumo[item.insumo] = 0;
        somaPorInsumo[item.insumo] += Number(item.totalKg || 0);
      });

      const entries = Object.entries(somaPorInsumo);
      if (entries.length > 0) {
        boxSoma.style.display = 'block';
        gridSoma.innerHTML = entries.map(([nome, kg]) => {
          const g = kg * 1000;
          const textoQtd = kg >= 1 ? `${kg.toFixed(3)} kg (${g.toFixed(0)} g)` : `${g.toFixed(1)} g (${kg.toFixed(4)} kg)`;
          return `<div class="sum-item">${esc(nome)}: <strong>${textoQtd}</strong></div>`;
        }).join('');
      } else {
        boxSoma.style.display = 'none';
      }
    } else {
      boxSoma.style.display = 'none';
    }
  }

  function esc(str) {
    return String(str)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }
</script>
